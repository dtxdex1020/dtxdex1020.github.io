handlers/call-ended.ts
on("call.ended", async (call) => {
  const intent = await classify(call.transcript);

  if (intent.kind === "book") {
    const slot = await calendar.firstOpen(intent.window);
    await sms.send(call.from, confirm(slot));
  }

  await owner.digest.add(summarize(call));
});
